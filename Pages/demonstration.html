Simple Demonstration

This section provides a basic demonstration of how the concept works. It is designed to show the core idea in a simple and understandable way, without requiring a full real-world deployment. The focus is on explaining how sound data moves through the system from detection to visualization.

Hardware Integration

The system utilizes the I2S protocol to ensure low-latency, high-fidelity audio capture. The hardware setup requires a stable 3.3V power connection and proper clock synchronization between the ESP32 master and the digital microphone slave.

                            #include <Arduino.h>
                            #include "driver/i2s.h"
                            #include <math.h>

                            #define I2S_WS 15
                            #define I2S_SD 32
                            #define I2S_SCK 14
                            #define SAMPLE_RATE 16000
                            #define BLOCK_SIZE 1024

                            void printTimestamp() {
                                unsigned long ms = millis() + 40815000;
                                unsigned long totalSeconds = ms / 1000;
                                unsigned long hours = (totalSeconds / 3600) % 24;
                                unsigned long minutes = (totalSeconds / 60) % 60;
                                unsigned long seconds = totalSeconds % 60;
                                unsigned long milliseconds = ms % 1000;

                                char timeStr[16];
                                snprintf(timeStr, sizeof(timeStr), "[%02lu:%02lu:%02lu.%03lu]", hours, minutes, seconds, milliseconds);
                                Serial.print(timeStr);
                            }

                            void setup() {
                                Serial.begin(115200);
                                while (!Serial) {
                                    delay(10);
                                }

                                printTimestamp();
                                Serial.println(" [INFO] Initializing I2S Bus...");

                                i2s_config_t i2s_config = {
                                    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
                                    .sample_rate = SAMPLE_RATE,
                                    .bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT,
                                    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
                                    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
                                    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
                                    .dma_buf_count = 8,
                                    .dma_buf_len = BLOCK_SIZE,
                                    .use_apll = false,
                                    .tx_desc_auto_clear = true,
                                    .fixed_mclk = 0
                                };

                                i2s_pin_config_t pin_config = {
                                    .bck_io_num = I2S_SCK,
                                    .ws_io_num = I2S_WS,
                                    .data_out_num = I2S_PIN_NO_CHANGE,
                                    .data_in_num = I2S_SD
                                };

                                esp_err_t err = i2s_driver_install(I2S_NUM_0, &i2s_config, 0, NULL);
                                if (err == ESP_OK) {
                                    i2s_set_pin(I2S_NUM_0, &pin_config);
                                    i2s_zero_dma_buffer(I2S_NUM_0);

                                    printTimestamp();
                                    Serial.println(" [SUCCESS] I2S Driver Installed.");
                                    printTimestamp();
                                    Serial.println(" [INFO] Configuration: 16000Hz / 32-bit");
                                } else {
                                    printTimestamp();
                                    Serial.print(" [ERROR] I2S driver install failed: ");
                                    Serial.println(err);
                                }
                            }

                            void loop() {
                                static int32_t buffer[BLOCK_SIZE];
                                size_t bytes_read = 0;

                                esp_err_t err = i2s_read(I2S_NUM_0, buffer, sizeof(buffer), &bytes_read, portMAX_DELAY);
                                if (err != ESP_OK) {
                                    printTimestamp();
                                    Serial.print(" [ERROR] I2S read failed: ");
                                    Serial.println(err);
                                    delay(1000);
                                    return;
                                }

                                int samples_read = bytes_read / sizeof(int32_t);
                                if (samples_read <= 0) {
                                    return;
                                }

                                double sum = 0.0;
                                for (int i = 0; i < samples_read; i++) {
                                    int32_t sample = buffer[i] >> 8; // convert 32-bit left-aligned to 24-bit signed
                                    sum += double(sample) * double(sample);
                                }

                                double rms = sqrt(sum / samples_read);
                                if (rms <= 0.0) {
                                    return;
                                }

                                double dbfs = 20.0 * log10(rms / 8388607.0);

                                printTimestamp();
                                Serial.print(" Raw RMS: ");
                                Serial.print(rms, 2);
                                Serial.print(" | dBFS: ");
                                Serial.print(dbfs, 2);
                                Serial.println();
                            }

Hardware Verification Checklist

Verify that your physical device behaves according to these core baseline performance guidelines during a standard testing evaluation:

  • Power Verification: The onboard red power LED lights up solid immediately when connected to a live USB interface port.
  • Data Transmission: The internal data processing routine refreshes automatically without causing device timeouts or reboots.
  • Acoustic Response: Making a sharp sound close to the hardware module shell causes a visible, corresponding rise in the calculated output properties.
